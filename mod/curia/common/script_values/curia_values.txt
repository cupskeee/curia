# Curia (M3 spike): script values used by the export. Numbers are read in localization with
# [EmptyScope.ScriptValue('name')|0] (any scope) or [THIS.ScriptValue('name')|0] (the current scope), the forms proven in M0/M0b.
# Rule from M0b: no saved scopes and no scope: references inside a script value (they come back empty and log errors);
# ROOT is never used. global_var: is not a scope reference.

# ---- opinion of the ruler ----
# Proven in M0b (vassal and heir scopes): a character's opinion of their liege.
curia_opinion_of_liege = {
	value = "opinion(liege)"
}
# The same string behind a guard, for the heir (who may have no liege): 0 when there is none.
# A variant of the proven form (wrapped in if/add), so it is not itself proven; curia_has_liege_value tells the cases apart.
curia_opinion_of_liege_guarded = {
	value = 0
	if = {
		limit = { exists = liege }
		add = "opinion(liege)"
	}
}
curia_has_liege_value = {
	value = 0
	if = {
		limit = { exists = liege }
		add = 1
	}
}

# ---- character flags ----
# 1 when the character holds land (is_landed is the trigger vanilla uses for "has any title with land").
curia_is_landed_value = {
	value = 0
	if = {
		limit = { is_landed = yes }
		add = 1
	}
}

# 1 independence, 2 liberty, 3 claimant, 4 populist, 5 peasant, 6 nomadic, 7 any other type (nation fracturing, escalated
# peasant and the ceremonial / imperial-policy types all give 7; identify those in the Factions window). 0 is never printed.
curia_probe_faction_type_code = {
	value = 0
	if = {
		limit = { faction_is_type = independence_faction }
		add = 1
	}
	else_if = {
		limit = { faction_is_type = liberty_faction }
		add = 2
	}
	else_if = {
		limit = { faction_is_type = claimant_faction }
		add = 3
	}
	else_if = {
		limit = { faction_is_type = populist_faction }
		add = 4
	}
	else_if = {
		limit = { faction_is_type = peasant_faction }
		add = 5
	}
	else_if = {
		limit = { faction_is_type = nomadic_faction }
		add = 6
	}
	else = {
		add = 7
	}
}
