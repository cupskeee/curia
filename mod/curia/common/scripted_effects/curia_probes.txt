# Curia (M3 spike): the probe section, run only by the "Curia probes" button (curia_probe_mode = 1), inside the frame
# between the last export section and END. Most lines are CURIA1|<snap>|PROBE|<name>|<value>; claim_detail has more
# fields (CURIA1|<snap>|PROBE|claim_detail|<index>|<title>|<4 flags>|<tier>).
# Each line tests ONE candidate form that was NOT proven in M0/M0b, so a failure is isolated to its own line: a wrong
# guess should print an empty value or 0, and the other lines still print.
#
# Turn probes off:
#   * everything: remove the line "curia_export_probes = yes" from curia_export_run (curia_export.txt);
#   * one group: remove its line from curia_export_probes below;
#   * one candidate: remove its change_global_variable + debug_log pair.
# Candidates that could have raised a script error rather than print an empty value (each is isolated in its own group).
# All of them ran in M3 session 2 (2026-10-11) and raised no entry in error.log; the groups stay to repeat the checks:
#   * claim_detail group: save_temporary_scope_as in the effect, scope:curia_probe_claimant = { has_*_claim_on = prev }
#     inside an ordered_claim body and its limit blocks (valid in the game's own effects; the M0b rule against scope:
#     only covers script values, which read the flags through global variables instead), and the explicit / pressed
#     parameters on ordered_claim (vanilla passes them to any_claim / every_claim / random_claim only);
#   * claim filter values (curia_probe_claims_e*_p*): explicit = no and seven of the nine explicit/pressed pairs appear in no
#     vanilla file, only in the trigger documentation; all nine pairs ran for a ruler and an heir;
#   * faction groups: faction_leader / faction_target links and joined_faction (all guarded with exists), and the
#     faction-scope script values (curia_probe_faction_*, faction_is_type, faction_power_threshold); they ran in three
#     factions (claimant, peasant, independence); the output is not capped (about 16 lines per faction);
#   * script values that iterate (every_vassal, every_vassal_or_below, every_claim, every_faction_member) and the
#     value-only triggers (monthly_character_income, yearly_character_income, realm_size, tier, faction_power,
#     faction_discontent): valid in the game's own files and ran without error.
# Still not run: an implicit claim flag at 1 (no ruler with an implicit claim was seen), a nonzero discontent, the other
# faction types, and the special_character claimant form (not in this file).
# A data function that does not exist would be reported when the localization is loaded (error.log); the names below are
# the ones the game's own GUI and localization files use.

curia_export_probes = {

# One line per claim (all claims, highest tier first, at most curia_max_claims), with the four claim kinds the game's
# triggers tell apart: explicit strong (pressed), explicit weak (unpressed), implicit strong, implicit weak.
# The ruler is saved as a scope so the title-scope iterator can ask the ruler about each title.
curia_probe_claim_detail = {
	save_temporary_scope_as = curia_probe_claimant
	set_global_variable = { name = curia_claim_index value = 0 }
	if = {
		limit = {
			any_claim = {
				explicit = all
				pressed = all
			}
		}
		ordered_claim = {
			explicit = all
			pressed = all
			max = curia_max_claims
			check_range_bounds = no
			order_by = tier
			change_global_variable = { name = curia_claim_index add = 1 }
			set_global_variable = { name = curia_claim_f_es value = 0 }
			set_global_variable = { name = curia_claim_f_ew value = 0 }
			set_global_variable = { name = curia_claim_f_is value = 0 }
			set_global_variable = { name = curia_claim_f_iw value = 0 }
			if = {
				limit = { scope:curia_probe_claimant = { has_strong_claim_on = prev } }
				set_global_variable = { name = curia_claim_f_es value = 1 }
			}
			if = {
				limit = { scope:curia_probe_claimant = { has_weak_claim_on = prev } }
				set_global_variable = { name = curia_claim_f_ew value = 1 }
			}
			if = {
				limit = { scope:curia_probe_claimant = { has_strong_implicit_claim_on = prev } }
				set_global_variable = { name = curia_claim_f_is value = 1 }
			}
			if = {
				limit = { scope:curia_probe_claimant = { has_weak_implicit_claim_on = prev } }
				set_global_variable = { name = curia_claim_f_iw value = 1 }
			}
			change_global_variable = { name = curia_export_lines add = 1 }
			debug_log = curia_probe_claim_detail
		}
	}
	else = {
		change_global_variable = { name = curia_export_lines add = 1 }
		debug_log = curia_probe_claim_detail_none
	}
}
